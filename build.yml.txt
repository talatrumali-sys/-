name: Build Android APK

on:
  push:
    branches: [main, master]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 20

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17

      - name: Install dependencies
        run: npm install

      - name: Create Android project
        run: npx cap add android

      - name: Apply app icon and version
        run: |
          cp -r resources/android/res/. android/app/src/main/res/
          sed -i "s/versionCode 1/versionCode ${{ github.run_number }}/" android/app/build.gradle
          sed -i "s/versionName \"1.0\"/versionName \"1.${{ github.run_number }}\"/" android/app/build.gradle

      - name: Sync web files
        run: npx cap sync android

      - name: Build installable debug APK
        working-directory: android
        run: |
          chmod +x gradlew
          ./gradlew assembleDebug --no-daemon
          cp app/build/outputs/apk/debug/app-debug.apk ../Tahaddi-Altawaqqoat.apk

      - uses: actions/upload-artifact@v4
        with:
          name: Tahaddi-Altawaqqoat-APK
          path: Tahaddi-Altawaqqoat.apk

      - name: Publish release (direct download link)
        uses: softprops/action-gh-release@v2
        with:
          tag_name: v${{ github.run_number }}
          name: تحدّي التوقعات v${{ github.run_number }}
          files: Tahaddi-Altawaqqoat.apk
